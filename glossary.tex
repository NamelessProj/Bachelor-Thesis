\newglossaryentry{opensource}{
    name={open source},
    description={Development model in which the source code is public and can be viewed, modified, and redistributed by anyone, subject to the terms of its license}
}

\newglossaryentry{js}{
    name={JavaScript},
    description={Programming language executed natively in web
browsers, used to make pages interactive and dynamic.
It forms the technical foundation of this project: \gls{reactjs}, the \gls{librairie}
used for the engine, is written in JavaScript, and the \acrshort{json} format is
directly interpretable by this language without any external dependency}
}

\newglossaryentry{ts}{
    name={TypeScript},
    description={Typed superset of \gls{js}, developed by Microsoft, which adds a static type system to the language. Types allow errors that would otherwise only surface at runtime in plain \gls{js} to be caught at compile time. TypeScript is compiled to standard \gls{js} before being executed in the browser \parencite{noauthor_starting_nodate}}
}

\newglossaryentry{reactjs}{
    name={React.js},
    description={\gls{js} library developed by Meta, used to build interactive web interfaces as reusable components. This project relies on React.js to render the \acrlong{vn} engine in the browser}
}

\newglossaryentry{frontend}{
    name={frontend},
    description={The part of a web application executed directly in the user's browser, responsible for display and interaction. Contrasts with the \glsit{backend}, which designates the server-side part}
}

\newglossaryentry{syntaxe}{
    name={syntax},
    description={The set of formal rules a file or script must follow to be correctly interpreted by a program. In this project, this term refers in particular to the structural rules of the \acrshort{json} format used as the engine's content source}
}

\newglossaryentry{parseur}{
    name={parser},
    description={Program or software component responsible for reading a text file and interpreting its structure according to defined rules. In this project, the parser analyses the scenario's \acrshort{json} file and checks that it conforms to the expected \gls{schema} before passing it on to the engine}
}

\newglossaryentry{schema}{
    name={schema},
    description={In computing, a formal description of the expected structure of a data file: which fields are required, which value types are accepted, and how elements are organised. In this project, the \acrshort{json} schema defines the structure that any scenario file must follow to be recognised and executed by the engine}
}

\newglossaryentry{librairie}{
    name={library},
    description={In software development, a set of reusable components or functions that a developer can integrate into their own project, as distributed software rather than a physical place}
}

\newglossaryentry{nocode}{
    name={no-code/low-code},
    description={Software development approaches aimed at reducing or
    removing the need to write code. No-code tools rely
    entirely on visual interfaces, while low-code tools
    allow code to be added on top for specific needs. These
    approaches aim to make application creation accessible to
    users without technical training}
}

\newglossaryentry{webassembly}{
    name={WebAssembly},
    description={Binary instruction format that allows code compiled from languages other than \gls{js} to run in a web browser, with near-native performance}
}

\newglossaryentry{backend}{
    name={backend},
    description={The server-side part of a web application, invisible to the end user, which handles business logic, data storage, and processing that requires a secure environment}
}

\newglossaryentry{npm}{
    name={npm},
    description={Package manager for \gls{js}, installed with Node.js. It allows reusable \glspl{librairie} (called \glsplit{package}) to be installed, updated, and published within a \gls{js} or \gls{ts} project. The \texttt{npm install} command downloads a project's dependencies, and \texttt{npm run build} runs a build script defined in the \texttt{package.json} file}
}

\newglossaryentry{package}{
    name={package},
    description={In \gls{js} development, a software module distributed via \gls{npm} and installable in a project through a single command. A package can be a \gls{librairie} (such as \gls{reactjs} or \gls{zod}), a \glsit{build} tool (such as \gls{vite}), or any other set of reusable features}
}

\newglossaryentry{webworker}{
    name={Web Worker},
    description={Native browser mechanism that allows \gls{js} code to run on a secondary execution thread, separate from the main thread that manages the user interface. This makes it possible to offload computationally expensive processing without blocking the display or the user's interactions. Web Workers communicate with the main thread by exchanging messages, without direct access to the \acrshort{dom} \parencite{noauthor_utilisation_2025}}
}

\newglossaryentry{plugin}{
    name={plugin},
    description={External software module that integrates with an existing tool to extend its functionality without modifying its source code}
}

\newglossaryentry{override}{
    name={override},
    description={In software development, a mechanism that allows a value or behaviour defined globally to be replaced locally, without modifying the global definition}
}

\newglossaryentry{tokenisation}{
    name={tokenization},
    description={The process of splitting a string of text into elementary units called \textit{tokens}}
}

\newglossaryentry{markdown}{
    name={Markdown},
    description={Lightweight markup language created by John Gruber in 2004, designed to be readable in its raw form while being convertible to \acrshort{html} \parencite{noauthor_daring_nodate}}
}

\newglossaryentry{framework}{
    name={framework},
    description={Software development framework providing an imposed structure and integrated tools for building an application. To be distinguished from a \gls{librairie}: a framework dictates the project's architecture, whereas a \gls{librairie} is integrated into an existing architecture at the developer's discretion}
}

\newglossaryentry{rendu}{
    name={render},
    description={In web interface development, the process by which the browser computes and updates a component's display in response to a change in state or data. In the context of \gls{reactjs}, a render specifically refers to the execution of a component's function and the recomputation of its virtual \acrshort{html} tree. An unnecessary render occurs when a component is recomputed even though its data has not changed}
}

\newglossaryentry{zod}{
    name={Zod},
    description={\Gls{schema} validation and type-inference \gls{librairie} for \gls{ts}, developed by Colin McDonnell \parencite{noauthor_zod_nodate}. It allows the expected structure of an object to be defined and data to be validated at runtime, producing precise error messages in case of non-conformance}
}

\newglossaryentry{hook}{
    name={hook},
    description={In \gls{reactjs}, a special function whose name begins with \texttt{use}, allowing a component's lifecycle features and state to be accessed from an ordinary \gls{js} function \parencite{noauthor_reference_nodate}}
}

\newglossaryentry{git}{
    name={Git},
    description={Distributed version control system created by Linus Torvalds in 2005 \parencite{noauthor_git_nodate}, used to track a project's change history, work on parallel branches, and collaborate on the same source code with several people}
}

\newglossaryentry{moscow}{
    name={MoSCoW},
    description={Requirements prioritisation method from agile project management. The acronym designates four priority levels: \textit{Must have}, \textit{Should have} (important but not blocking), \textit{Could have} (desirable if time allows), and \textit{Won't have} (out of scope for this iteration) \parencite{clegg_case_1994}}
}

\newglossaryentry{pdca}{
    name={PDCA},
    description={Four-phase continuous improvement cycle: \textit{Plan}, \textit{Do} (implementation), \textit{Check} (evaluation), and \textit{Act} (adjustment). Formalised by W. Edwards Deming \parencite{deming_out_1986}}
}

\newglossaryentry{vite}{
    name={Vite},
    description={\Glsit{frontend} \glsit{build} tool and development server created by Evan You, designed to offer fast hot-reloading and optimised compilation for modern \gls{js} projects \parencite{noauthor_vite_nodate}}
}

\newglossaryentry{poc}{
    name={proof of concept},
    description={Demonstration or functional prototype intended to verify the technical feasibility of an idea or approach, without necessarily covering every use case or reaching production-level polish}
}

\newglossaryentry{snaphtml}{
    name={HTML snapshot},
    description={In the context of the \texttt{Typewriter} component, a string representing the complete \acrshort{html} \gls{rendu} state at a given moment of the typewriter animation. Each snapshot corresponds to a precise number of visible displayed characters, with \acrshort{html} tags correctly opened and closed, enabling O(1) display per animation tick (Author 2026)}
}

\newglossaryentry{sprite}{
    name={sprite},
    description={Image or illustration representing a character or graphical element, displayed over a scene's background and positionable or animatable independently of it. Term originating from two-dimensional video game development}
}

\newglossaryentry{memoisation}{
    name={memoization},
    description={Optimisation technique consisting of caching the result of an expensive computation or function call, so that it can be reused on subsequent calls with the same parameters instead of being recomputed}
}

\newglossaryentry{runtime}{
    name={runtime},
    description={A program's execution environment, as opposed to its compilation. Also refers, by extension, to the software component responsible for interpreting and executing content at that point, for example the engine that interprets a scenario file in the browser}
}

\newglossaryentry{placeholder}{
    name={placeholder},
    description={Temporary text or value displayed in an empty input field, indicating to the user the kind of content expected, and disappearing as soon as they begin entering a value}
}

\newglossaryentry{build}{
    name={build},
    description={Process of compiling and assembling a project's source code into optimised files ready to be deployed. In this project, this step is executed by the \texttt{npm run build} command}
}

\newglossaryentry{negativetesting}{
    name={negative testing},
    description={(\textit{unhappy path testing}) Testing approach aimed at verifying a program's behaviour when faced with unexpected, invalid, or erroneous input, as opposed to \textit{positive testing}, which covers expected behaviour under normal conditions. Hora defines negative tests as those that verify ``abnormal execution, such as exceptional cases''\footnote{French original: «~l'exécution anormale, comme les cas exceptionnels~»} \parencite{hora_test_2024}}
}

\newglossaryentry{reactprofiler}{
    name={React DevTools Profiler},
    description={Browser extension developed by Meta Open Source that records and analyses the rendering performance of a \gls{reactjs} application, in particular the duration and frequency of per-component \gls{rendu} cycles \parencite{noauthor_react_nodate}}
}
